dfs = []
for r in rounds:
    df = pd.read_csv(r, sep='\t', index_col=0)
    df = df[['ESVAR']]
    name = r.split("/")[-1].replace(".csv", "")
    df.index = df.index + "_%s"%name
    dfs.append(df)
    
df = pd.concat(dfs)
